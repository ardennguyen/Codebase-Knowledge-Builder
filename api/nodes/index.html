    nodes

    File Path: nodes.py

    Following the orchestration logic defined in main and DAG construction in flow, the nodes module provides the concrete computational components (nodes) that execute each pipeline stage of Codebase-Knowledge-Builder. Built on top of pocketflow.Node and pocketflow.BatchNode, these classes handle repository file extraction, context routing and batch chunking, LLM abstraction identification, relationship mapping, chapter sequencing, content generation, and multi-format output compilation.

    def parse_yaml_response(response):
    """Extract and parse YAML from an LLM response fenced in ```yaml blocks."""
    try:

    create_token_counter()

    Visibility: Public
    Signature: def create_token_counter() -> Callable[[str], int]:

    Description:
    Instantiates a token-counting closure using tiktoken's "cl100k_base" encoding, with an automatic fallback to an approximate character-based calculation (len(text) // 4) if tiktoken fails.

    def create_token_counter():
    """Create a token counting function using tiktoken with char-count fallback."""
    try:
        enc = tiktoken.get_encoding("cl100k_base")
        return lambda text: len(enc.encode(text, disallowed_special=()))
    except Exception:
        return lambda text: len(text) // 4

    The closure disables special token checks (disallowed_special=()) so arbitrary source code containing tokens like <|endoftext|> can be processed without runtime exceptions.

    Parameters: * None

    Returns: * Callable[[str], int]: Function that takes a string and returns estimated or exact token count.

    Raises: * None (exceptions during tokenizer initialization trigger fallback).

    Example: 

    count_tokens = create_token_counter()
entry_tokens = count_tokens(entry)

    resolve_max_tokens()

    Visibility: Public
    Signature: def resolve_max_tokens(shared: dict) -> int:

    Description:
    Determines the maximum token window for the active LLM by inspecting the shared state or looking up model configurations across supported providers (Gemini, OpenAI, Anthropic, Ollama, etc.) via get_model_context_length in utils.call_llm.

    def resolve_max_tokens(shared):
    """Resolve max_tokens from shared store or auto-detect from provider env vars."""
    max_tokens = shared.get("max_tokens")
    if max_tokens is not None:
        return max_tokens
    provider = os.environ.get("LLM_PROVIDER")
    if provider == "GEMINI" or not provider:
        endpoint = "https://generativelanguage.googleapis.com"
        model_name = os.getenv("GEMINI_MODEL", "gemini-3.7-flash")
        api_key = os.getenv("GEMINI_API_KEY", "")
    else:
        endpoint = os.environ.get(f"{provider}_BASE_URL", "")
        model_name = os.environ.get(f"{provider}_MODEL", "")
        api_key = os.environ.get(f"{provider}_API_KEY", "")
    return get_model_context_length(endpoint, model_name, api_key)

    post()

    Visibility: Public
    Signature: def post(self, shared: dict, prep_res: tuple, exec_res: list[int]) -> str:

    Description:
    Generates 1:1 module abstraction descriptors and sets sequential chapter ordering for all retained code files.

    def post(self, shared, prep_res, exec_res):
    import os
    files = shared.get("files", [])
    valid_indices = set(exec_res)
    modules = []
    chapter_order = []

    for idx, (file_path, content) in enumerate(files):
        if idx not in valid_indices:
            print(f"  - Skipping non-code file: {file_path}")
            continue

        clean_name = os.path.splitext(file_path)[0].replace(os.sep, ".").replace("/", ".")

        modules.append({
            "name": clean_name,
            "description": f"Internal API reference for `{file_path}`",
            "files": [idx],
            "original_path": file_path
        })
        chapter_order.append(len(modules) - 1)

    shared["abstractions"] = modules
    shared["chapter_order"] = chapter_order
    shared["relationships"] = {"summary": "Deterministic Internal API Reference.", "details": []}
    print(f"\033[92m[DeterministicFileMapper] Mapped {len(modules)} ACTUAL code files for exhaustive documentation.\033[0m")
    return "default"

    Parameters: * shared (dict): Pipeline shared state. * prep_res (tuple): The parameters produced in prep(). * exec_res (list[int]): List of valid file indices from exec().

    Returns: * str: Default route transition identifier ("default").

    ContextRouter

    Inherits from: pocketflow.Node

    Analyzes repository file sizes against the LLM context limit and decides whether the execution DAG follows the "direct", "batch", or "deterministic" pathway.

    _build_directory_tree()

    Visibility: Protected (Static)
    Signature: def _build_directory_tree(files_data: list[tuple[str, str]]) -> str:

    Description:
    Generates a structured, compact directory tree string from file tuples with embedded index annotations.

    @staticmethod
def _build_directory_tree(files_data):
    """Build a compact directory tree string from the list of (path, content) tuples."""
    from collections import defaultdict
    dir_files = defaultdict(list)
    for i, (path, _content) in enumerate(files_data):
        dirname = os.path.dirname(path) or "."
        basename = os.path.basename(path)
        dir_files[dirname].append(f"{basename} (idx:{i})")

    lines = []
    for dirname in sorted(dir_files.keys()):
        lines.append(f"{dirname}/")
        for fname in sorted(dir_files[dirname]):
            lines.append(f"  {fname}")
    return "\n".join(lines)

    Parameters: * files_data (list[tuple[str, str]]): List of (path, content) tuples.

    Returns: * str: Compact indented directory tree.

    exec()

    Visibility: Public
    Signature: def exec(self, prep_res: tuple) -> str | list[list[tuple[int, str, str]]]:

    Description:
    If the batch route is chosen, partitions files into token-bounded batches grouped strictly by directory to preserve architectural coherence.

    def exec(self, prep_res):
    route, files_data, effective_limit, batch_size, file_token_map, count_tokens, directory_tree, debug = prep_res
    llm_logger.info(f"NODE EXEC | node=ContextRouter | action=route_decision | route={route} | files={len(files_data)} | effective_limit={effective_limit:,}")

    if route == "direct":
        return "direct"
    if route == "deterministic":
        return "deterministic"

    # Group by directory for coherence, with pre-computed tokens
    dir_groups = defaultdict(list)
    for i, (path, content) in enumerate(files_data):
        dir_groups[os.path.dirname(path)].append((i, path, content, file_token_map[i]))

    batches = []
    for dirname in sorted(dir_groups.keys()):
        current_batch = []
        current_tokens = 0
        for i, path, content, tokens in dir_groups[dirname]:
            if current_batch and (current_tokens + tokens > effective_limit or len(current_batch) >= batch_size):
                batches.append(current_batch)
                current_batch = []
                current_tokens = 0
            current_batch.append((i, path, content))
            current_tokens += tokens
        if current_batch:
            batches.append(current_batch)

    self._directory_tree = directory_tree
    return batches

    Parameters: * prep_res (tuple): Unpacked routing parameters from prep().

    Returns: * str | list[list[tuple[int, str, str]]]: Either route string ("direct" / "deterministic") or list of partitioned file batches.

    Inherits from: pocketflow.Node

    Evaluates inter-abstraction relationships, generates dependency links with descriptive labels, and produces an overarching project summary using a two-pass token redistribution algorithm.

    prep()

    Visibility: Public
    Signature: def prep(self, shared: dict) -> tuple:

    Description:
    Calculates token budgets per abstraction, applies a two-pass algorithm to distribute available token headroom evenly while reallocating unused budget to larger abstractions, and builds the relationship prompt context.

    def prep(self, shared):
    abstractions = shared["abstractions"]
    files_data = shared["files"]
    # ... [Build initial context and calculate token budgets] ...
    per_abstr_budget = total_budget // max(num_abstractions, 1)
    included_indices = set()
    abstr_results = []  # list of (included_files, remaining_files, unused_budget)

    # Pass 1: Give each abstraction an equal share, track unused
    for i, sized in enumerate(abstr_file_data):
        budget = per_abstr_budget
        included_files, remaining_files = [], []
        for idx, path, file_content, tokens in sized:
            if idx in included_indices:
                included_files.append((idx, path, None, 0))
                continue
            if tokens <= budget:
                included_files.append((idx, path, file_content, tokens))
                budget -= tokens
                included_indices.add(idx)
            else:
                remaining_files.append((idx, path, file_content, tokens))
        abstr_results.append((included_files, remaining_files, budget))

    # Pass 2: Redistribute unused budget to abstractions with remaining files
    total_unused = sum(r[2] for r in abstr_results)
    if total_unused > 0:
        for i, (included_files, remaining_files, _unused) in enumerate(abstr_results):
            if not remaining_files or total_unused <= 0:
                continue
            still_remaining = []
            for idx, path, file_content, tokens in remaining_files:
                if tokens <= total_unused:
                    included_files.append((idx, path, file_content, tokens))
                    total_unused -= tokens
                    included_indices.add(idx)
                else:
                    still_remaining.append((idx, path, file_content, tokens))
            abstr_results[i] = (included_files, still_remaining, 0)
    # ... [Assemble context strings] ...
    return (context, "\n".join(abstraction_info_for_prompt), num_abstractions, ...)
