    nodes

    File Path: nodes.py

    Overview

    The nodes module defines the execution units that make up the computational directed acyclic graph (DAG) orchestrated by flow and invoked via main. Built upon the pocketflow pipeline framework (Node and BatchNode), each class implements the lifecycle pattern prep -> exec -> post.

    These nodes are responsible for fetching repository assets, performing dynamic context window calculation and token budgeting, mapping architectural abstractions directly or via batch map-reduce passes, analyzing cross-module dependencies, scheduling chapter order, generating Markdown content, handling incremental MD5 cache invalidation, and aggregating final tutorial and MkDocs documentation outputs.

    classDiagram
    class Node {
        +prep(shared)
        +exec(prep_res)
        +post(shared, prep_res, exec_res)
    }
    class BatchNode {
        +prep(shared)
        +exec(item)
        +post(shared, prep_res, exec_res_list)
    }

    get_content_for_indices()

    Visibility: Public
    Signature: def get_content_for_indices(files_data: list, indices: list) -> dict:

    Description:
    Extracts file contents from a list of (path, content) tuples for a specific subset of numerical indices. Constructs a dictionary keyed by formatted index strings ("{i} # {path}") to provide traceable context for LLM prompt generation.

    def get_content_for_indices(files_data, indices):
    content_map = {}
    for i in indices:
        if 0 <= i < len(files_data):
            path, content = files_data[i]
            content_map[f"{i} # {path}"] = (
                content  # Use index + path as key for context
            )
    return content_map

    Parameters: * files_data (list): List of tuples containing (path, content). * indices (list): Target file indices to extract.

    Returns: * dict: Mapping of string keys ("index # path") to file text contents.

    Raises: * None directly raised; handles out-of-bounds index lookups safely via bounds checking.

    Example: 

    # Extracting specific file contents based on abstraction indices
related_file_indices = [0, 2]
files_data = [("app.py", "print(1)"), ("util.py", "x = 2"), ("main.py", "run()")]
content_map = get_content_for_indices(files_data, related_file_indices)

    parse_yaml_response()

    Visibility: Public
    Signature: def parse_yaml_response(response: str) -> Any:

    Description:
    Extracts YAML code blocks delimited by triple backticks (```yaml) from raw LLM responses and deserializes them into Python native data structures via yaml.safe_load.

    def parse_yaml_response(response):
    """Extract and parse YAML from an LLM response fenced in ```yaml blocks."""
    try:
        yaml_str = response.strip().split("```yaml")[1].split("```")[0].strip()
        return yaml.safe_load(yaml_str)
    except Exception as e:
        raise ValueError(f"Failed to parse YAML: {e}")

    create_token_counter()

    Visibility: Public
    Signature: def create_token_counter() -> Callable[[str], int]:

    Description:
    Instantiates and returns a tokenizer counting function. Attempts to load the tiktoken BPE encoding cl100k_base. If unavailable, falls back to a 4-characters-per-token heuristic calculation.

    def create_token_counter():
    """Create a token counting function using tiktoken with char-count fallback."""
    try:
        enc = tiktoken.get_encoding("cl100k_base")
        return lambda text: len(enc.encode(text, disallowed_special=()))
    except Exception:
        return lambda text: len(text) // 4

    Parameters: * None

    Returns: * Callable[[str], int]: A callable function taking a string and returning its estimated token length.

    Raises: * None directly raised.

    Example: 

    count_tokens = create_token_counter()
total_tokens = count_tokens("sample text")

    resolve_max_tokens()

    Visibility: Public
    Signature: def resolve_max_tokens(shared: dict) -> int:

    Description:
    Resolves the context window token limit. Reads directly from shared["max_tokens"] if defined; otherwise inspects environment variables (LLM_PROVIDER, endpoint URLs, API keys) and queries get_model_context_length().

    def resolve_max_tokens(shared):
    """Resolve max_tokens from shared store or auto-detect from provider env vars."""
    max_tokens = shared.get("max_tokens")
    if max_tokens is not None:
        return max_tokens
    provider = os.environ.get("LLM_PROVIDER")
    if provider == "GEMINI" or not provider:
        endpoint = "https://generativelanguage.googleapis.com"
        model_name = os.getenv("GEMINI_MODEL", "gemini-3.7-flash")
        api_key = os.getenv("GEMINI_API_KEY", "")
    else:
        endpoint = os.environ.get(f"{provider}_BASE_URL", "")
        model_name = os.environ.get(f"{provider}_MODEL", "")
        api_key = os.environ.get(f"{provider}_API_KEY", "")
    return get_model_context_length(endpoint, model_name, api_key)

    Parameters: * shared (dict): The global pipeline shared context dictionary.

    Returns: * int: Maximum context length in tokens.

    Raises: * None directly raised.

    Example: 

    max_tokens = resolve_max_tokens(shared)

    Inherits from: pocketflow.Node

    Filters non-code files (UI layouts, build scripts, configuration metadata) using LLM evaluation and generates direct 1:1 module-to-chapter abstractions without dynamic clustering. Used predominantly in api-reference mode.

    Visibility: Public
    Signature: def prep(self, shared: dict) -> tuple:

    Description:
    Extracts indexed codebase file listings from shared["files"] and prepares an LLM prompt requesting identification of actual source code files containing business logic, APIs, or classes.

        def prep(self, shared):
        files_data = shared["files"]
        project_name = shared["project_name"]

        file_listing = "\n".join([f"{i} # {path}" for i, (path, _) in enumerate(files_data)])

        prompt = f"""For the project `{project_name}`, here is the list of all files in the codebase:

{file_listing}

Your task is to identify WHICH of these files are ACTUAL CODE files that contain APIs, functions, classes, or core business logic.
EXCLUDE: UI layouts (like .xaml, .storyboard, .html), configuration files (like .xml, .json, .manifest, .ini), static assets, build scripts (like .csproj, .sln), and documentation.

Return ONLY a YAML list of the file indices that should be documented as code modules.

```yaml
- 0
- 1
- 3
```"""
        return prompt, shared.get("thinking_level", None), shared.get("max_tokens", 100000)

    Returns: * tuple: (prompt: str, thinking_level: Any, max_tokens: int).

    DeterministicFileMapper.post()

    Visibility: Public
    Signature: def post(self, shared: dict, prep_res: tuple, exec_res: list) -> str:

    Description:
    Constructs module structures from validated indices, populates shared["abstractions"], sets shared["chapter_order"] sequentially, and initializes baseline relationship metadata.

        def post(self, shared, prep_res, exec_res):
        import os
        files = shared.get("files", [])
        valid_indices = set(exec_res)
        modules = []
        chapter_order = []

        for idx, (file_path, content) in enumerate(files):
            if idx not in valid_indices:
                print(f"  - Skipping non-code file: {file_path}")
                continue

            clean_name = os.path.splitext(file_path)[0].replace(os.sep, ".").replace("/", ".")

            modules.append({
                "name": clean_name,
                "description": f"Internal API reference for `{file_path}`",
                "files": [idx],
                "original_path": file_path
            })
            chapter_order.append(len(modules) - 1)

        shared["abstractions"] = modules
        shared["chapter_order"] = chapter_order
        shared["relationships"] = {"summary": "Deterministic Internal API Reference.", "details": []}
        print(f"\033[92m[DeterministicFileMapper] Mapped {len(modules)} ACTUAL code files for exhaustive documentation.\033[0m")
        return "default"

    Parameters: * shared (dict): Shared state pipeline dictionary. * prep_res (tuple): Results from prep(). * exec_res (list): List of valid file indices from exec().

    Returns: * str: Router action identifier ("default").

    ContextRouter

    Inherits from: pocketflow.Node

    Performs token calculation and budgeting against the LLM's context window. Selects the execution route: "deterministic" (for API reference), "direct" (when files fit in single context), or "batch" (when map-reduce is required).

    ContextRouter.prep()

    Visibility: Public
    Signature: def prep(self, shared: dict) -> tuple:

    Description:
    Calculates directory tree overhead and prompt template token limits using create_token_counter(). Determines whether codebase token usage exceeds the effective safety margin (95% of max context limit minus prompt overhead) or if batching was explicitly forced.

        def prep(self, shared):
        files_data = shared["files"]
        max_tokens = resolve_max_tokens(shared)

        shared["max_tokens"] = max_tokens

        # --- Token estimation setup ---
        count_tokens = create_token_counter()

        # --- Calculate prompt overhead FIRST ---
        # 1. Prompt template (measure both, use the larger)
        prompt_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), "prompts")
        max_template_tokens = 0
        for subdir in ["tutorial", "advanced"]:
            template_path = os.path.join(prompt_dir, subdir, "map_abstractions.md")
            if os.path.exists(template_path):
                with open(template_path, "r", encoding="utf-8-sig") as f:
                    t = count_tokens(f.read())
                max_template_tokens = max(max_template_tokens, t)

        # 2. Directory tree (built from all files, shared across batches)
        directory_tree = self._build_directory_tree(files_data)
        tree_tokens = count_tokens(directory_tree)

        prompt_overhead = max_template_tokens + tree_tokens
        # ... token mapping calculations ...
        safety_limit = int(max_tokens * 0.95)
        effective_limit = safety_limit - prompt_overhead
        force_batch = shared.get("force_batch", False)

        if shared.get("mode", "tutorial") == "api-reference":
            return ("deterministic", files_data, effective_limit, shared.get("batch_size", 50),
                    None, None, directory_tree, False)

        if total_tokens > effective_limit or force_batch:
            return ("batch", files_data, effective_limit, shared.get("batch_size", 50),
                    file_token_map, count_tokens, directory_tree, shared.get("debug", False))

        return ("direct", files_data, effective_limit, shared.get("batch_size", 50),
                None, None, directory_tree, False)

    Parameters: * shared (dict): Pipeline shared state dictionary.

    Returns: * tuple: Routing descriptor tuple containing target branch strategy and token budgeting context.

    ContextRouter.exec()

    Visibility: Public
    Signature: def exec(self, prep_res: tuple) -> Union[str, list]:

    Description:
    If the selected route is "direct" or "deterministic", passes through the route string. In "batch" mode, partitions codebase files by directory into token-budgeted and size-capped batches without crossing directory boundaries.

        def exec(self, prep_res):
        route, files_data, effective_limit, batch_size, file_token_map, count_tokens, directory_tree, debug = prep_res
        if route == "direct":
            return "direct"
        if route == "deterministic":
            return "deterministic"

        # Group by directory for coherence, with pre-computed tokens
        dir_groups = defaultdict(list)
        for i, (path, content) in enumerate(files_data):
            dir_groups[os.path.dirname(path)].append((i, path, content, file_token_map[i]))

        # Build token-aware batches (never mix directories)
        batches = []
        for dirname in sorted(dir_groups.keys()):
            current_batch = []
            current_tokens = 0

            for i, path, content, tokens in dir_groups[dirname]:
                if current_batch and (current_tokens + tokens > effective_limit or len(current_batch) >= batch_size):
                    batches.append(current_batch)
                    current_batch = []
                    current_tokens = 0

                current_batch.append((i, path, content))
                current_tokens += tokens

            if current_batch:
                batches.append(current_batch)

        self._directory_tree = directory_tree
        return batches

    Parameters: * prep_res (tuple): Prepared execution parameters from prep().

    Returns: * Union[str, list]: The string route identifier ("direct", "deterministic") or a nested list of file batches.

    ContextRouter._build_directory_tree()

    Visibility: Private
    Signature: def _build_directory_tree(files_data: list) -> str:

    Description:
    Static helper method that constructs an indented plain-text representation of the codebase's directory structure, tagging each file entry with its index number.

        @staticmethod
    def _build_directory_tree(files_data):
        """Build a compact directory tree string from the list of (path, content) tuples."""
        from collections import defaultdict
        dir_files = defaultdict(list)
        for i, (path, _content) in enumerate(files_data):
            dirname = os.path.dirname(path) or "."
            basename = os.path.basename(path)
            dir_files[dirname].append(f"{basename} (idx:{i})")

        lines = []
        for dirname in sorted(dir_files.keys()):
            lines.append(f"{dirname}/")
            for fname in sorted(dir_files[dirname]):
                lines.append(f"  {fname}")
        return "\n".join(lines)

    Parameters: * files_data (list): List of (path, content) tuples.

    Returns: * str: Compact, formatted directory hierarchy.

    MapAbstractions.exec()

    Visibility: Public
    Signature: def exec(self, item: dict) -> list:

    Description:
    Formats the map_abstractions prompt template with batch contents and directory tree, executes the LLM call via call_llm(), and parses and validates abstraction dictionaries.

        def exec(self, item):
        batch_index = item["batch_index"]
        files = item["files"]
        # ... prompt construction ...
        prompt_template = load_prompt_template("map_abstractions", mode=item.get("mode", "tutorial"))
        # ... template formatting ...
        response = call_llm(prompt, use_cache=(item["use_cache"] and self.cur_retry == 0), thinking_level=item["thinking_level"])

        abstractions = parse_yaml_response(response)

        validated_abstractions = []
        if isinstance(abstractions, list):
            for obj in abstractions:
                if isinstance(obj, dict) and "name" in obj and "description" in obj and "file_indices" in obj:
                    import re
                    validated_indices = []
                    for idx_entry in obj["file_indices"]:
                        nums = re.findall(r'\d+', str(idx_entry))
                        if nums:
                            validated_indices.append(int(nums[0]))
                    if validated_indices:
                        validated_abstractions.append({
                            "name": obj["name"],
                            "description": obj["description"],
                            "files": sorted(list(set(validated_indices)))
                        })
        return validated_abstractions
